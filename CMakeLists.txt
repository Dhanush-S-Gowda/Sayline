cmake_minimum_required(VERSION 3.21)
project(Sayline VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Network DBus)
option(SAYLINE_CORE_ONLY "Build protocol tests without the desktop dependencies" OFF)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(protocol-test tests/protocol_test.cpp src/transcript.cpp src/pastekeys.cpp)
    target_include_directories(protocol-test PRIVATE src)
    target_link_libraries(protocol-test PRIVATE Qt6::Core Qt6::Test)
    add_test(NAME protocol COMMAND protocol-test)
    add_executable(runtime-test tests/runtime_test.cpp src/runtime.cpp)
    target_include_directories(runtime-test PRIVATE src)
    target_link_libraries(runtime-test PRIVATE Qt6::Core Qt6::Test)
    add_test(NAME runtime-storage COMMAND runtime-test)
    add_executable(clipboard-test tests/clipboard_test.cpp src/clipboardwriter.cpp src/clipboardwriter.h)
    target_include_directories(clipboard-test PRIVATE src)
    target_link_libraries(clipboard-test PRIVATE Qt6::Core Qt6::Test)
    add_test(NAME verified-clipboard COMMAND clipboard-test)
    set_tests_properties(verified-clipboard PROPERTIES TIMEOUT 10)
    add_test(NAME installer-preflight COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/installer_test.sh")
    find_program(NODE_EXECUTABLE node)
    if(NODE_EXECUTABLE)
        add_test(NAME kwin-script COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/kwin_test.cjs")
    endif()
endif()
if(NOT SAYLINE_CORE_ONLY)
    find_package(Qt6 6.7 REQUIRED COMPONENTS Quick QuickControls2 Multimedia WebSockets)
    qt_add_executable(sayline src/main.cpp src/controller.cpp src/controller.h
        src/transcript.cpp src/transcript.h src/pastekeys.cpp src/pastekeys.h src/runtime.cpp src/runtime.h src/clipboardwriter.cpp src/clipboardwriter.h)
    qt_add_resources(sayline overlay PREFIX "/" FILES qml/Main.qml qml/icons/punctuation.svg packaging/focus.js sound.mp3 sayline-icon.ico)
    target_link_libraries(sayline PRIVATE Qt6::Gui Qt6::Quick Qt6::QuickControls2
        Qt6::Network Qt6::DBus Qt6::Multimedia Qt6::WebSockets)
    if(BUILD_TESTING)
        add_executable(controller-test tests/controller_test.cpp src/controller.cpp
            src/controller.h src/transcript.cpp src/pastekeys.cpp src/runtime.cpp src/clipboardwriter.cpp src/clipboardwriter.h)
        target_include_directories(controller-test PRIVATE src)
        target_link_libraries(controller-test PRIVATE Qt6::Test Qt6::Gui Qt6::Network
            Qt6::DBus Qt6::Multimedia Qt6::WebSockets)
        add_executable(overlay-test tests/overlay_test.cpp)
        target_link_libraries(overlay-test PRIVATE Qt6::Test Qt6::Quick Qt6::QuickControls2)
        target_compile_definitions(overlay-test PRIVATE SAYLINE_BINARY_DIR="${CMAKE_CURRENT_BINARY_DIR}")
        qt_add_resources(overlay-test overlay_test_resources PREFIX "/" FILES qml/Main.qml qml/icons/punctuation.svg)
        add_test(NAME overlay-hover-click COMMAND overlay-test)
        set_tests_properties(overlay-hover-click PROPERTIES TIMEOUT 10
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
        add_executable(bridge-engine-test tests/bridge_engine_test.cpp)
        target_link_libraries(bridge-engine-test PRIVATE Qt6::Test Qt6::Qml)
        target_compile_definitions(bridge-engine-test PRIVATE SAYLINE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
        add_test(NAME qt-kwin-bridge COMMAND bridge-engine-test)
        set_tests_properties(qt-kwin-bridge PROPERTIES TIMEOUT 10)
        add_test(NAME controller-lifecycle COMMAND controller-test)
        add_test(NAME overlay-startup-exit COMMAND sayline --smoke-test)
        set_tests_properties(overlay-startup-exit PROPERTIES TIMEOUT 10
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;XDG_STATE_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-state")
    endif()
    install(TARGETS sayline RUNTIME DESTINATION bin)
    install(FILES packaging/org.sayline.Sayline.desktop DESTINATION share/applications)
    install(FILES packaging/icons/org.sayline.Sayline.png DESTINATION share/icons/hicolor/256x256/apps)
endif()

if(NOT SAYLINE_CORE_ONLY)
    install(DIRECTORY packaging/kwin-shortcut/ DESTINATION share/kwin/scripts/sayline-shortcut)
    # Resolve the prefix at INSTALL time, so --prefix and DESTDIR are respected.
    install(CODE [[
        set(service_dir "$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/share/dbus-1/services")
        file(MAKE_DIRECTORY "${service_dir}")
        set(executable "${CMAKE_INSTALL_PREFIX}/bin/sayline")
        string(REPLACE "\\" "\\\\" executable "${executable}")
        string(REPLACE "\"" "\\\"" executable "${executable}")
        set(unit_dir "$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/share/systemd/user")
        file(MAKE_DIRECTORY "${unit_dir}")
        file(WRITE "${unit_dir}/sayline-history-cleanup.service"
            "[Unit]\nDescription=Remove Sayline transcripts older than 24 hours\n[Service]\nType=oneshot\nExecStart=\"${executable}\" --prune-history\n")
        file(WRITE "${unit_dir}/sayline-history-cleanup.timer"
            "[Unit]\nDescription=Hourly Sayline transcript retention check\n[Timer]\nOnCalendar=hourly\nPersistent=true\n[Install]\nWantedBy=timers.target\n")
        file(WRITE "${service_dir}/org.sayline.Sayline.service"
            "[D-BUS Service]\nName=org.sayline.Sayline\nExec=\"${executable}\" --background\n")
    ]])
endif()
